add_executable(odl-plugin-host host.cpp)
if(NOT WIN32)
  target_link_libraries(odl-plugin-host PRIVATE ${CMAKE_DL_LIBS})
endif()
foreach(revision IN ITEMS 1 2)
  add_library(reloadable-plugin-v${revision} SHARED sample_plugin.cpp)
  target_compile_definitions(reloadable-plugin-v${revision} PRIVATE ODL_SAMPLE_REVISION=${revision})
  set_target_properties(reloadable-plugin-v${revision} PROPERTIES PREFIX "")
endforeach()
install(TARGETS odl-plugin-host reloadable-plugin-v1 reloadable-plugin-v2 RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(FILES plugin_api.h DESTINATION include)
add_library(refuse-stop-plugin SHARED sample_plugin.cpp)
target_compile_definitions(refuse-stop-plugin PRIVATE ODL_SAMPLE_REFUSE_STOP=1)
set_target_properties(refuse-stop-plugin PROPERTIES PREFIX "")
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    set(control_client)
    if(WIN32)
      list(APPEND control_client --client $<TARGET_FILE:open-dll-loader>)
    endif()
    add_test(NAME cooperative_reload COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_reload.py"
      --host $<TARGET_FILE:odl-plugin-host> --v1 $<TARGET_FILE:reloadable-plugin-v1>
      --v2 $<TARGET_FILE:reloadable-plugin-v2> --refuse $<TARGET_FILE:refuse-stop-plugin> ${control_client})
  endif()
endif()
