cmake_minimum_required(VERSION 3.24)
project(OpenDllLoader VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
if(NOT WIN32)
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
  add_subdirectory(examples/reloadable-host)
  message(STATUS "Native Linux build: cooperative plugin host/example only; the desktop loader is Win32.")
  return()
endif()
if(NOT MSVC)
  message(FATAL_ERROR "The Windows desktop build requires MSVC.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /WX /permissive- /utf-8 /guard:cf)
add_link_options(/DYNAMICBASE /NXCOMPAT /guard:cf /Brepro)
add_subdirectory(examples/reloadable-host)
add_library(loader STATIC src/loader.cpp src/managed_host.cpp)
target_include_directories(loader PUBLIC src)
target_link_libraries(loader PUBLIC bcrypt)
add_executable(open-dll-loader src/cli.cpp)
target_link_libraries(open-dll-loader PRIVATE loader)
add_executable(OpenDllLoader WIN32 src/gui.cpp src/app.manifest)
target_link_libraries(OpenDllLoader PRIVATE loader comctl32 comdlg32 shell32 uxtheme gdiplus)
add_executable(odl-demo-launcher WIN32 src/demo_launcher.cpp src/app.manifest)
add_library(test-plugin SHARED tests/plugin.cpp)
if(BUILD_TESTING)
  add_test(NAME own_process_loading COMMAND open-dll-loader --self-test $<TARGET_FILE:test-plugin>)
  add_test(NAME native_ui_layout COMMAND OpenDllLoader --ui-test "${CMAKE_CURRENT_BINARY_DIR}/ui-preview.png")
endif()
install(TARGETS OpenDllLoader open-dll-loader RUNTIME DESTINATION .)
install(TARGETS odl-demo-launcher RUNTIME DESTINATION .)
